7. Bulk bootstrap script

#!/usr/bin/env bash
set -euo pipefail
mkdir -p vendor
repos=(
  iden3/circom iden3/snarkjs iden3/circomlib
  iden3/circom-witnesscalc arkworks-rs/circom-compat
  flwrlabs/flower
)
for r in "${repos[@]}"; name
  name="${r##*/}"
  gh repo view "$r" --json url >/dev/null
  if [[ ! -d "vendor/$name/.git" ]]; then
    gh repo clone "$r" "vendor/$name" -- --depth 1
  fi
  git -C "vendor/$name" rev-parse HEAD > "vendor/$name.pin"
done
echo "Vendor pins written."
